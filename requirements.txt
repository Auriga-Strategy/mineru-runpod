# Container-side dependencies. Consumers do NOT need these — they only need
# the mineru_client subpackage which has its own pyproject.toml.

# MinerU with vLLM backend support. PGB pins the exact patch release because a
# benchmark image must not change when rebuilt. The 3.4.x default VLM is
# MinerU2.5-Pro-2605-1.2B and the pipeline OCR is PP-OCRv6; both model repos
# are pinned to immutable revisions in the Dockerfile. A version bump must
# update and re-test parser plus both baked revisions together. MinerU 3.4.5
# requires vllm >=0.10.1.1,<0.22.0, satisfied by the Dockerfile FROM tag.
mineru[core,vllm]==3.4.5

# RunPod serverless SDK — provides runpod.serverless.start() and async helpers.
runpod==1.12.0

# The engine-agnostic half of this worker: input transport (url / b64 /
# volume_path + format detection), outbound target checks, response packaging
# (tarball / inline / s3), structured logging, redaction and the model-cache
# probe. This repo keeps the MinerU engine, the input schema and the job
# envelope; everything above comes from here.
#
# Pinned to a tag, not a branch: the harness API is pre-release and expected to
# move, and an image build has to be reproducible. The archive URL is a PEP 508
# direct reference, so the build node needs no `git` binary. Kept in sync with
# the `test` extra in pyproject.toml, which is what CI installs.
#
# The `[s3]` extra is boto3, imported lazily inside the s3 packaging path — a
# worker without S3 env vars never touches it. Drop the extra on a fork that
# only returns tarballs or inline payloads.
runpod-doc-worker[s3] @ https://github.com/sergeyshmakov/runpod-doc-worker/archive/refs/tags/v0.5.0.tar.gz

# Used by the handler for downloading URL-supplied PDFs.
httpx==0.28.1

# Xet storage backend client used by huggingface_hub for chunk-based
# parallel downloads. Auto-detected when installed; the Dockerfile bake
# step sets HF_XET_HIGH_PERFORMANCE=1 to saturate bandwidth and CPU
# during the model snapshot pull. Replaces the deprecated hf_transfer.
# Runtime stays offline so this is build-time only.
hf-xet==1.6.0

# OpenTelemetry stack. Dormant unless OTEL_EXPORTER_OTLP_ENDPOINT is set
# in the worker environment — worker/telemetry.py reads that env var and
# only imports the SDK / configures exporters when it is non-empty. So
# these packages add ~30 MB to the image but cost zero overhead at
# runtime for the default no-telemetry path. See docs/.../observability.mdx.
# Forks that never want OTel can drop these four lines without other code
# changes.
opentelemetry-api==1.44.0
opentelemetry-sdk==1.44.0
opentelemetry-exporter-otlp-proto-http==1.44.0
# nvidia-ml-py is the upstream-maintained successor to pynvml; we keep
# the `import pynvml` call shape since both packages register the same
# top-level module. Pinned explicitly here so the Dockerfile install
# layer doesn't depend on vllm's transitive pull. Powers the
# mineru.gpu.* gauges in worker/telemetry.py.
nvidia-ml-py==13.610.43
